    $tagstack  = array();
    $stacksize = 0;

    $single_tags = array( 'area', 'base', 'basefont', 'br', 'col',
        'command', 'embed', 'frame', 'hr', 'img', 'input', 'isindex',
        'link', 'meta', 'param', 'source', 'track', 'wbr' );
